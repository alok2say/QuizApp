**Goal:** Let a logged-in user see a flat list of quizzes, start one, answer
questions against a countdown timer with answers cached in `localStorage`,
submit once (manually or on timer expiry), get server-computed exact-match
scoring, and view an end-of-attempt result breakdown.

**Architecture:** Next.js App Router server components read data straight
from Prisma (no API route needed for reads). Two API routes handle the
only two mutations — starting an attempt and submitting one — both
delegating to plain TypeScript functions in `src/lib/quiz/` so the scoring
and retake-eligibility logic is unit-testable without HTTP or Next.js
plumbing. Client components are used only where the browser is required:
the timer and the `localStorage` answer cache on the take-quiz page.

**Tech Stack:** Next.js 15 App Router, TypeScript, Prisma 7 (`prisma`
client already generated from `prisma/schema.prisma`), NextAuth v4
(Credentials + JWT, already wired via `src/lib/auth.ts`), Vitest.

**Spec:** `CONTEXT.md` (session continuity log) + `prisma/schema.prisma`
(committed data model, ground truth for fields/enums). `SPEC.md`'s
narrative MCQ section was lost during Task 1's `create-next-app` file
shuffle; the design decisions below were re-confirmed with the user this
session and should be treated as authoritative going forward — copy them
into `SPEC.md` once this plan is executed (see "Post-plan spec repair"
at the bottom).

## Global Constraints

- Feedback mode in scope: `END_ONLY` only. `IMMEDIATE` mode is explicitly
  out of scope (per `CONTEXT.md`) and must not be implemented here.
- Scoring is server-side, exact-match, no partial credit: a question is
  correct only if the submitted `selectedOptionIds` set is exactly equal
  to the set of options where `isCorrect: true` — this rule is the same
  for `SINGLE_SELECT` and `MULTI_SELECT`, so one scoring function covers
  both.
- Retake enforcement: `SINGLE_ATTEMPT` blocks starting a second attempt —
  if the existing attempt is unsubmitted, the user resumes it; if
  submitted, they're redirected to its result. `UNLIMITED_BEST` and
  `UNLIMITED_LATEST` always allow a new `Attempt` row to be created.
- `isLeaderboardEligible` is a denormalized flag recomputed on every
  submit across all of that user's attempts for that quiz: `SINGLE_ATTEMPT`
  → the one attempt is eligible; `UNLIMITED_LATEST` → only the
  most-recently-submitted attempt is eligible; `UNLIMITED_BEST` → only the
  highest-scoring attempt is eligible (tie-break: earliest `submittedAt`
  wins, for determinism).
- An `Attempt` row is created when the user starts the quiz
  (`submittedAt: null`) and updated in place on submit — not created fresh
  at submit time.
- Timer expiry auto-submits whatever is cached in `localStorage` at that
  moment; unanswered questions score 0.
- All new server-side logic that isn't purely UI-rendering must have a
  Vitest unit test written and run **before** the implementation, per this
  project's established TDD flow (see `docs/LEARNING_LOG.md`).

```typescript
// src/lib/quiz/scoring.ts
export type ScorableQuestion = {
  id: string
  points: number
  options: { id: string; isCorrect: boolean }[]
}

export type ScoredAnswer = {
  questionId: string
  isCorrect: boolean
  pointsAwarded: number
}

export function scoreAnswer(question: ScorableQuestion, selectedOptionIds: string[]): ScoredAnswer {
  const correctIds = question.options.filter((o) => o.isCorrect).map((o) => o.id)
  const selected = new Set(selectedOptionIds)
  const correct = new Set(correctIds)

  const isCorrect =
    selected.size === correct.size && [...selected].every((id) => correct.has(id))

  return {
    questionId: question.id,
    isCorrect,
    pointsAwarded: isCorrect ? question.points : 0,
  }
}
```

```typescript
// src/lib/quiz/attempts.ts
import { prisma } from '@/lib/prisma'

export type StartAttemptResult =
  | { ok: true; attemptId: string }
  | { ok: false; status: 404 | 409; error: string; attemptId?: string; resume?: boolean }

export async function startAttempt(quizId: string, userId: string): Promise<StartAttemptResult> {
  const quiz = await prisma.quiz.findUnique({ where: { id: quizId } })
  if (!quiz) {
    return { ok: false, status: 404, error: 'Quiz not found.' }
  }

  if (quiz.retakePolicy === 'SINGLE_ATTEMPT') {
    const existing = await prisma.attempt.findMany({ where: { quizId, userId } })
    if (existing.length > 0) {
      const attempt = existing[0]
      return {
        ok: false,
        status: 409,
        error: 'You have already attempted this quiz.',
        attemptId: attempt.id,
        resume: attempt.submittedAt === null,
      }
    }
  }

  const attempt = await prisma.attempt.create({ data: { quizId, userId } })
  return { ok: true, attemptId: attempt.id }
}
```

```typescript
// src/app/api/quizzes/[quizId]/start/route.ts
import { getCurrentUser } from '@/lib/session'
import { startAttempt } from '@/lib/quiz/attempts'

export async function POST(
  _request: Request,
  { params }: { params: Promise<{ quizId: string }> },
) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: 'Not authenticated.' }, { status: 401 })
  }

  const { quizId } = await params
  const result = await startAttempt(quizId, user.id)

  if (!result.ok) {
    return Response.json(
      { error: result.error, attemptId: result.attemptId, resume: result.resume },
      { status: result.status },
    )
  }

  return Response.json({ attemptId: result.attemptId }, { status: 200 })
}
```

```typescript
// append to src/lib/quiz/attempts.ts
import { scoreAnswer } from './scoring'

export type SubmitAnswerInput = { questionId: string; selectedOptionIds: string[] }

export type SubmitAttemptResult =
  | { ok: true; score: number; maxScore: number }
  | { ok: false; status: 403 | 404 | 409; error: string }

export async function submitAttempt(
  attemptId: string,
  userId: string,
  answers: SubmitAnswerInput[],
): Promise<SubmitAttemptResult> {
  const attempt = await prisma.attempt.findUnique({
    where: { id: attemptId },
    include: { quiz: { include: { questions: { include: { options: true } } } } },
  })

  if (!attempt) {
    return { ok: false, status: 404, error: 'Attempt not found.' }
  }
  if (attempt.userId !== userId) {
    return { ok: false, status: 403, error: 'Forbidden.' }
  }
  if (attempt.submittedAt !== null) {
    return { ok: false, status: 409, error: 'Attempt already submitted.' }
  }

  const answersByQuestion = new Map(answers.map((a) => [a.questionId, a.selectedOptionIds]))

  const scored = attempt.quiz.questions.map((question) =>
    scoreAnswer(question, answersByQuestion.get(question.id) ?? []),
  )

  const score = scored.reduce((sum, s) => sum + s.pointsAwarded, 0)
  const maxScore = attempt.quiz.questions.reduce((sum, q) => sum + q.points, 0)
  const submittedAt = new Date()
  const completionSeconds = Math.round((submittedAt.getTime() - attempt.startedAt.getTime()) / 1000)

  await prisma.attemptAnswer.createMany({
    data: scored.map((s) => ({
      attemptId,
      questionId: s.questionId,
      selectedOptionIds: answersByQuestion.get(s.questionId) ?? [],
      isCorrect: s.isCorrect,
      pointsAwarded: s.pointsAwarded,
    })),
  })

  const siblings = await prisma.attempt.findMany({
    where: { quizId: attempt.quizId, userId },
  })

  const isCurrentEligible = await resolveLeaderboardEligibility(
    attempt.quiz.retakePolicy,
    siblings.map((a) => ({
      id: a.id,
      score: a.id === attemptId ? score : a.score,
      submittedAt: a.id === attemptId ? submittedAt : a.submittedAt,
    })),
    attemptId,
  )

  await prisma.attempt.update({
    where: { id: attemptId },
    data: { score, submittedAt, completionSeconds, isLeaderboardEligible: isCurrentEligible },
  })

  return { ok: true, score, maxScore }
}

// Recomputes isLeaderboardEligible across every attempt this user has for
// this quiz. Returns whether the attempt being submitted right now ends up
// eligible, so the caller can fold that into its own single update instead
// of a second unconditional write that would race with this function's.
async function resolveLeaderboardEligibility(
  retakePolicy: string,
  attempts: { id: string; score: number; submittedAt: Date | null }[],
  currentAttemptId: string,
): Promise<boolean> {
  let eligibleId: string

  if (retakePolicy === 'UNLIMITED_BEST') {
    eligibleId = attempts.reduce((best, a) => {
      if (a.score > best.score) return a
      if (a.score === best.score && (a.submittedAt?.getTime() ?? Infinity) < (best.submittedAt?.getTime() ?? Infinity)) return a
      return best
    }).id
  } else {
    // SINGLE_ATTEMPT and UNLIMITED_LATEST: the attempt being submitted now wins.
    eligibleId = currentAttemptId
  }

  for (const a of attempts) {
    if (a.id === currentAttemptId) continue
    await prisma.attempt.update({
      where: { id: a.id },
      data: { isLeaderboardEligible: a.id === eligibleId },
    })
  }

  return eligibleId === currentAttemptId
}
```

```typescript
import { z } from 'zod'
import { getCurrentUser } from '@/lib/session'
import { submitAttempt } from '@/lib/quiz/attempts'

const submitSchema = z.object({
  answers: z.array(
    z.object({
      questionId: z.string(),
      selectedOptionIds: z.array(z.string()),
    }),
  ),
})

export async function POST(
  request: Request,
  { params }: { params: Promise<{ attemptId: string }> },
) {
  const user = await getCurrentUser()
  if (!user) {
    return Response.json({ error: 'Not authenticated.' }, { status: 401 })
  }

  const json = await request.json()
  const parsed = submitSchema.safeParse(json)
  if (!parsed.success) {
    return Response.json({ error: 'Invalid submission body.' }, { status: 400 })
  }

  const { attemptId } = await params
  const result = await submitAttempt(attemptId, user.id, parsed.data.answers)

  if (!result.ok) {
    return Response.json({ error: result.error }, { status: result.status })
  }

  return Response.json({ score: result.score, maxScore: result.maxScore }, { status: 200 })
}
```

```typescript
// src/lib/quiz/queries.ts
import { prisma } from '@/lib/prisma'

export type QuizListItem = {
  id: string
  title: string
  description: string | null
  timeLimitSeconds: number
  retakePolicy: 'UNLIMITED_BEST' | 'UNLIMITED_LATEST' | 'SINGLE_ATTEMPT'
  latestAttempt: { id: string; score: number; submittedAt: Date | null } | null
}

export async function getQuizzesWithStatus(userId: string): Promise<QuizListItem[]> {
  const quizzes = await prisma.quiz.findMany({
    include: {
      attempts: {
        where: { userId },
        orderBy: { startedAt: 'desc' },
        take: 1,
      },
    },
  })

  return quizzes.map((quiz) => ({
    id: quiz.id,
    title: quiz.title,
    description: quiz.description,
    timeLimitSeconds: quiz.timeLimitSeconds,
    retakePolicy: quiz.retakePolicy,
    latestAttempt: quiz.attempts[0]
      ? {
          id: quiz.attempts[0].id,
          score: quiz.attempts[0].score,
          submittedAt: quiz.attempts[0].submittedAt,
        }
      : null,
  }))
}
```

```typescript
import { useState } from 'react'

export function StartQuizButton({ quizId, label }: { quizId: string; label: string }) {
  const router = useRouter()
  const [pending, setPending] = useState(false)
  const [error, setError] = useState<string | null>(null)

  async function handleClick() {
    setPending(true)
    setError(null)

    const res = await fetch(`/api/quizzes/${quizId}/start`, { method: 'POST' })
    const body = await res.json()

    setPending(false)

    if (res.ok) {
      router.push(`/quizzes/${quizId}/take/${body.attemptId}`)
      return
    }

    if (res.status === 409 && body.attemptId) {
      const destination = body.resume
        ? `/quizzes/${quizId}/take/${body.attemptId}`
        : `/quizzes/${quizId}/result/${body.attemptId}`
      router.push(destination)
      return
    }

    setError(body.error ?? 'Could not start the quiz.')
  }

  return (
    <div>
      <button onClick={handleClick} disabled={pending}>
        {pending ? 'Starting...' : label}
      </button>
      {error && <p role="alert">{error}</p>}
    </div>
  )
}
```

```typescript
export type TakeableQuestion = {
  id: string
  text: string
  order: number
  type: 'SINGLE_SELECT' | 'MULTI_SELECT'
  options: { id: string; text: string }[]
}

export type QuizForTaking =
  | { ok: true; quizTitle: string; timeLimitSeconds: number; startedAt: Date; questions: TakeableQuestion[] }
  | { ok: false; status: 403 | 404 | 409; error: string }

export async function getQuizForTaking(
  quizId: string,
  attemptId: string,
  userId: string,
): Promise<QuizForTaking> {
  const attempt = await prisma.attempt.findUnique({
    where: { id: attemptId },
    include: { quiz: { include: { questions: { include: { options: true } } } } },
  })

  if (!attempt) {
    return { ok: false, status: 404, error: 'Attempt not found.' }
  }
  if (attempt.userId !== userId) {
    return { ok: false, status: 403, error: 'Forbidden.' }
  }
  if (attempt.submittedAt !== null) {
    return { ok: false, status: 409, error: 'This attempt was already submitted.' }
  }

  const questions = [...attempt.quiz.questions]
    .sort((a, b) => a.order - b.order)
    .map((q) => ({
      id: q.id,
      text: q.text,
      order: q.order,
      type: q.type,
      options: q.options.map((o) => ({ id: o.id, text: o.text })),
    }))

  return {
    ok: true,
    quizTitle: attempt.quiz.title,
    timeLimitSeconds: attempt.quiz.timeLimitSeconds,
    startedAt: attempt.startedAt,
    questions,
  }
}
```

```typescript
'use client'

import { useEffect, useMemo, useRef, useState } from 'react'
import { useRouter } from 'next/navigation'
import type { TakeableQuestion } from '@/lib/quiz/queries'

type Props = {
  quizId: string
  attemptId: string
  quizTitle: string
  questions: TakeableQuestion[]
  remainingSeconds: number
}

type AnswersMap = Record<string, string[]>

function storageKey(attemptId: string) {
  return `quiz-attempt-${attemptId}`
}

export function QuizTakeForm({ quizId, attemptId, quizTitle, questions, remainingSeconds }: Props) {
  const router = useRouter()
  const [answers, setAnswers] = useState<AnswersMap>({})
  const [secondsLeft, setSecondsLeft] = useState(remainingSeconds)
  const [submitting, setSubmitting] = useState(false)
  const submittedRef = useRef(false)

  useEffect(() => {
    const cached = localStorage.getItem(storageKey(attemptId))
    if (cached) setAnswers(JSON.parse(cached))
  }, [attemptId])

  useEffect(() => {
    localStorage.setItem(storageKey(attemptId), JSON.stringify(answers))
  }, [answers, attemptId])

  const submit = useMemo(
    () => async () => {
      if (submittedRef.current) return
      submittedRef.current = true
      setSubmitting(true)

      const payload = {
        answers: questions.map((q) => ({
          questionId: q.id,
          selectedOptionIds: answers[q.id] ?? [],
        })),
      }

      const res = await fetch(`/api/attempts/${attemptId}/submit`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      })

      if (res.ok) {
        localStorage.removeItem(storageKey(attemptId))
        router.push(`/quizzes/${quizId}/result/${attemptId}`)
      } else {
        setSubmitting(false)
        submittedRef.current = false
      }
    },
    [answers, attemptId, quizId, questions, router],
  )

  useEffect(() => {
    if (secondsLeft <= 0) {
      submit()
      return
    }
    const timer = setTimeout(() => setSecondsLeft((s) => s - 1), 1000)
    return () => clearTimeout(timer)
  }, [secondsLeft, submit])

  function toggleOption(question: TakeableQuestion, optionId: string) {
    setAnswers((prev) => {
      const current = prev[question.id] ?? []
      if (question.type === 'SINGLE_SELECT') {
        return { ...prev, [question.id]: [optionId] }
      }
      const next = current.includes(optionId)
        ? current.filter((id) => id !== optionId)
        : [...current, optionId]
      return { ...prev, [question.id]: next }
    })
  }

  return (
    <main>
      <h1>{quizTitle}</h1>
      <p>Time left: {secondsLeft}s</p>

      {questions.map((question) => (
        <fieldset key={question.id}>
          <legend>{question.text}</legend>
          {question.options.map((option) => (
            <label key={option.id}>
              <input
                type={question.type === 'SINGLE_SELECT' ? 'radio' : 'checkbox'}
                name={question.id}
                checked={(answers[question.id] ?? []).includes(option.id)}
                onChange={() => toggleOption(question, option.id)}
              />
              {option.text}
            </label>
          ))}
        </fieldset>
      ))}

      <button onClick={submit} disabled={submitting}>
        {submitting ? 'Submitting...' : 'Submit Quiz'}
      </button>
    </main>
  )
}
```

```typescript
// append to src/lib/quiz/queries.ts
export type AnsweredQuestion = {
  id: string
  text: string
  isCorrect: boolean
  pointsAwarded: number
  points: number
  options: { id: string; text: string; isCorrect: boolean; wasSelected: boolean }[]
}

export type AttemptResult =
  | { ok: true; quizTitle: string; score: number; maxScore: number; questions: AnsweredQuestion[] }
  | { ok: false; status: 403 | 404 | 409; error: string }

export async function getAttemptResult(attemptId: string, userId: string): Promise<AttemptResult> {
  const attempt = await prisma.attempt.findUnique({
    where: { id: attemptId },
    include: {
      quiz: { include: { questions: { include: { options: true } } } },
      answers: true,
    },
  })

  if (!attempt) {
    return { ok: false, status: 404, error: 'Attempt not found.' }
  }
  if (attempt.userId !== userId) {
    return { ok: false, status: 403, error: 'Forbidden.' }
  }
  if (attempt.submittedAt === null) {
    return { ok: false, status: 409, error: 'This attempt has not been submitted yet.' }
  }

  const answersByQuestion = new Map(attempt.answers.map((a) => [a.questionId, a]))
  const maxScore = attempt.quiz.questions.reduce((sum, q) => sum + q.points, 0)

  const questions = attempt.quiz.questions.map((question) => {
    const answer = answersByQuestion.get(question.id)
    const selected = new Set(answer?.selectedOptionIds ?? [])

    return {
      id: question.id,
      text: question.text,
      isCorrect: answer?.isCorrect ?? false,
      pointsAwarded: answer?.pointsAwarded ?? 0,
      points: question.points,
      options: question.options.map((o) => ({
        id: o.id,
        text: o.text,
        isCorrect: o.isCorrect,
        wasSelected: selected.has(o.id),
      })),
    }
  })

  return {
    ok: true,
    quizTitle: attempt.quiz.title,
    score: attempt.score,
    maxScore,
    questions,
  }
}
```
